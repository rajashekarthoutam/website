module lfsr_16bit_root_of_trust (
    input  wire clk_27mhz,
    input  wire rst_n,
    input  wire uart_rx_trigger,
    output reg  [15:0] rolling_code,
    output reg  lock_status
);
    // Tang Nano 20K Hardware Root of Trust
    always @(posedge clk_27mhz or negedge rst_n) begin
        if (!rst_n)
            rolling_code <= 16'hACE1;
        else begin
            // LFSR Polynomial: x^16 + x^14 + x^13 + x^11 + 1
            rolling_code <= {rolling_code[14:0], 
                rolling_code[15] ^ rolling_code[13] ^ 
                rolling_code[12] ^ rolling_code[10]};
        end
    end
endmodule
